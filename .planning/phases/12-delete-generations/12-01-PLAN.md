   ```sql
   ADD COLUMN deleted_at TIMESTAMPTZ NULL;
   ```

2. Add partial index for efficient filtering of non-deleted records:
   ```sql
   CREATE INDEX idx_generations_not_deleted
   ON generations(job_id)
   WHERE deleted_at IS NULL;
   ```

3. Create or replace the atomic counter decrement function:
   ```sql
   CREATE OR REPLACE FUNCTION decrement_job_generation_count(
     p_job_id UUID
   ) RETURNS void AS $$
   BEGIN
     UPDATE jobs
     SET completed_generations = GREATEST(completed_generations - 1, 0)
     WHERE id = p_job_id;
   END;
   $$ LANGUAGE plpgsql;
   ```

4. Grant execute permission to authenticated users:
   ```sql
   GRANT EXECUTE ON FUNCTION decrement_job_generation_count(UUID) TO authenticated;
   ```
