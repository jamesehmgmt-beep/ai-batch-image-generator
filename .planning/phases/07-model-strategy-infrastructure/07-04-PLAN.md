```typescript
import { ModelId, ModelStrategy } from './types';
import { NanoBananaStrategy } from './nano-banana-strategy';
import { SeedreamStrategy } from './seedream-strategy';

/**
 * Factory function to get the appropriate model strategy
 *
 * Usage:
 *   const strategy = getModelStrategy('nano-banana-pro');
 *   await strategy.createTask(params);
 *
 * Strategies are created fresh per call (stateless, no caching needed).
 * This enables parallel usage without concurrency issues.
 *
 * @param modelId - The model identifier
 * @returns ModelStrategy implementation for the requested model
 * @throws Error if modelId is unknown (with exhaustive check for type safety)
 */
export function getModelStrategy(modelId: ModelId): ModelStrategy {
  switch (modelId) {
    case 'nano-banana-pro':
      return new NanoBananaStrategy();

    case 'seedream-4.5-edit':
      return new SeedreamStrategy();

    default:
      // TypeScript exhaustive check - if we add a new ModelId,
      // this will cause a compile error until we handle it
      const exhaustiveCheck: never = modelId;
      throw new Error(`Unknown model: ${exhaustiveCheck}`);
  }
}

/**
 * Get model strategy safely, returning undefined for invalid IDs
 * Useful when validating user input before creating strategy
 */
export function tryGetModelStrategy(modelId: string): ModelStrategy | undefined {
  try {
    return getModelStrategy(modelId as ModelId);
  } catch {
    return undefined;
  }
}
```

The exhaustive check pattern ensures adding new models to ModelId type will cause compile-time errors until the factory is updated.
