   - UNKNOWN_ERROR - Catch-all for unclassified errors

  <action>
Create a new file `lib/queue/retry-strategies.ts` with:

1. `classifyError(errorMessage?: string): ErrorClassification` function:
   - Parse error message string to determine error type
   - Pattern matching:
     - Contains '401' or 'unauthorized' -> AUTH_ERROR, not retryable
     - Contains '402' or 'payment' -> AUTH_ERROR, not retryable
     - Contains '422' or 'validation' or 'invalid' -> VALIDATION_ERROR, not retryable
     - Contains '429' or 'rate limit' -> RATE_LIMIT, retryable with exponential
     - Contains '5' followed by two digits (5xx) -> SERVER_ERROR, retryable with exponential
     - Contains 'timeout' or 'ECONNREFUSED' -> NETWORK_ERROR, retryable with exponential
     - Default -> UNKNOWN_ERROR, retryable with exponential (safer to retry)
   - Return appropriate userMessage for each type:
     - AUTH: "API authentication failed - check KIE_API_KEY"
     - PAYMENT: "API payment required - check kie.ai account balance"
     - VALIDATION: "Invalid request parameters - check image URLs"
     - RATE_LIMIT: "Rate limited, waiting before retry..."
     - SERVER: "Server error, retrying..."
     - NETWORK: "Network error, retrying..."
     - UNKNOWN: "Generation failed, retrying..."

2. `calculateBackoff(attemptNumber: number, strategy: RetryStrategy): number` function:
   - If strategy is 'immediate': return 0
   - If strategy is 'fixed': return 3000 (3 seconds)
   - If strategy is 'exponential':
     - Base delay: 1000ms (1 second)
     - Max delay: 60000ms (60 seconds cap)
     - Factor: 2 (doubles each attempt)
     - Calculate: baseDelay * Math.pow(factor, attemptNumber - 1)
     - Cap at maxDelay
     - Apply FULL JITTER: Math.random() * exponentialDelay
     - Return Math.floor(jitteredDelay)

3. `sleep(ms: number): Promise<void>` utility function:
   - Simple promise-based sleep for use in retry loops

Import ErrorClassification and RetryStrategy from '../types/errors'.
Export all functions.

```typescript
import { describe, it, expect } from 'vitest';
import { classifyError, calculateBackoff, sleep } from '../retry-strategies';

describe('classifyError', () => {
  it('classifies 401 errors as non-retryable auth errors', () => {
    const result = classifyError('Non-retryable error (401): Unauthorized');
    expect(result.retryable).toBe(false);
    expect(result.strategy).toBe('immediate');
    expect(result.userMessage).toContain('authentication');
  });

  it('classifies 402 errors as non-retryable payment errors', () => {
    const result = classifyError('Non-retryable error (402): Payment required');
    expect(result.retryable).toBe(false);
    expect(result.strategy).toBe('immediate');
  });

  it('classifies 422 errors as non-retryable validation errors', () => {
    const result = classifyError('Non-retryable error (422): Invalid input');
    expect(result.retryable).toBe(false);
    expect(result.strategy).toBe('immediate');
  });

  it('classifies 429 errors as retryable rate limits', () => {
    const result = classifyError('Retryable error (429): Too Many Requests');
    expect(result.retryable).toBe(true);
    expect(result.strategy).toBe('exponential');
    expect(result.userMessage).toContain('Rate limited');
  });

  it('classifies 5xx errors as retryable server errors', () => {
    const result = classifyError('Retryable error (503): Service Unavailable');
    expect(result.retryable).toBe(true);
    expect(result.strategy).toBe('exponential');
  });

  it('classifies network errors as retryable', () => {
    const result = classifyError('ECONNREFUSED');
    expect(result.retryable).toBe(true);
  });

  it('defaults to retryable for unknown errors', () => {
    const result = classifyError('Something weird happened');
    expect(result.retryable).toBe(true);
    expect(result.strategy).toBe('exponential');
  });

  it('handles undefined error message', () => {
    const result = classifyError(undefined);
    expect(result.retryable).toBe(true);
  });
});

describe('calculateBackoff', () => {
  it('returns 0 for immediate strategy', () => {
    expect(calculateBackoff(1, 'immediate')).toBe(0);
  });

  it('returns 3000 for fixed strategy', () => {
    expect(calculateBackoff(1, 'fixed')).toBe(3000);
    expect(calculateBackoff(5, 'fixed')).toBe(3000);
  });

  it('returns exponential values with jitter', () => {
    // Run multiple times to verify jitter
    const delays = Array.from({ length: 10 }, () =>
      calculateBackoff(3, 'exponential')
    );

    // All should be different (jitter)
    const uniqueDelays = new Set(delays);
    expect(uniqueDelays.size).toBeGreaterThan(1);

    // All should be <= 4000 (base * 2^2 = 1000 * 4)
    delays.forEach(d => expect(d).toBeLessThanOrEqual(4000));
  });

  it('caps at 60 seconds', () => {
    // Attempt 10: 1000 * 2^9 = 512000, should cap at 60000
    const delays = Array.from({ length: 100 }, () =>
      calculateBackoff(10, 'exponential')
    );
    delays.forEach(d => expect(d).toBeLessThanOrEqual(60000));
  });
});

describe('sleep', () => {
  it('waits for specified duration', async () => {
    const start = Date.now();
    await sleep(50);
    const elapsed = Date.now() - start;
    expect(elapsed).toBeGreaterThanOrEqual(45); // Allow small variance
  });
});
```
