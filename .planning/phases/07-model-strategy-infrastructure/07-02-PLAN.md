import pRetry, { AbortError } from 'p-retry';
import {
  ModelStrategy,
  ModelCapabilities,
  ModelGenerationParams,
  NanoBananaParams,
  NANO_BANANA_CAPABILITIES,
} from './types';

/**
 * Strategy implementation for Nano Banana Pro model
 * Supports: resolution (1K/2K/4K), output_format (png/jpg), max 8 reference images
 */
export class NanoBananaStrategy implements ModelStrategy {
  readonly capabilities: ModelCapabilities = NANO_BANANA_CAPABILITIES;

  /**
   * Validate Nano Banana Pro specific parameters
   * Throws descriptive error if validation fails
   */
  validateParams(params: ModelGenerationParams & Partial<NanoBananaParams>): void {
    // Validate resolution
    const resolution = (params as NanoBananaParams).resolution;
    if (!resolution || !['1K', '2K', '4K'].includes(resolution)) {
      throw new Error(
        `Nano Banana Pro requires resolution: 1K, 2K, or 4K. Got: ${resolution}`
      );
    }

    // Validate reference image count
    if (params.referenceImages.length > this.capabilities.maxReferenceImages) {
      throw new Error(
        `Nano Banana Pro supports max ${this.capabilities.maxReferenceImages} reference images. Got: ${params.referenceImages.length}`
      );
    }

    // Validate aspect ratio
    if (!this.capabilities.supportedAspectRatios.includes(params.aspectRatio)) {
      throw new Error(
        `Nano Banana Pro does not support aspect ratio: ${params.aspectRatio}. Supported: ${this.capabilities.supportedAspectRatios.join(', ')}`
      );
    }
  }

  /**
   * Create a kie.ai task for Nano Banana Pro generation
   * Uses p-retry for automatic retry with exponential backoff
   */
  async createTask(params: NanoBananaParams): Promise<string> {
    this.validateParams(params);

    const apiKey = this.getApiKey();
    const payload = {
      model: 'nano-banana-pro' as const,
      input: {
        prompt: params.prompt,
        image_input: params.referenceImages.slice(0, this.capabilities.maxReferenceImages),
        aspect_ratio: params.aspectRatio,
        resolution: params.resolution,
        output_format: params.outputFormat.toLowerCase() as 'png' | 'jpg',
      },
    };

    return pRetry(
      async () => {
        const response = await fetch('https://api.kie.ai/api/v1/jobs/createTask', {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${apiKey}`,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify(payload),
        });

        // Non-retryable errors
        if ([401, 402, 422].includes(response.status)) {
          const errorText = await response.text();
          throw new AbortError(`Non-retryable error (${response.status}): ${errorText}`);
        }

        // Retryable errors
        if (response.status === 429 || response.status >= 500) {
          throw new Error(`Retryable error (${response.status}): ${response.statusText}`);
        }

        if (!response.ok) {
          const errorText = await response.text();
          throw new AbortError(`API error (${response.status}): ${errorText}`);
        }

        const data = await response.json();
        if (!data.data?.taskId) {
          throw new AbortError(`Invalid response: missing taskId`);
        }

        return data.data.taskId;
      },
      {
        retries: 5,
        factor: 2,
        minTimeout: 1000,
        maxTimeout: 30000,
        randomize: true,
        onFailedAttempt: (error) => {
          console.log(`[NanoBanana] createTask attempt ${error.attemptNumber} failed. ${error.retriesLeft} retries left.`);
        },
      }
    );
  }

   */
  async pollTask(taskId: string): Promise<{ resultUrl: string }> {

    const apiKey = process.env.KIE_API_KEY;

      throw new Error('KIE_API_KEY environment variable is required');
    }
    return apiKey;
  }

/**
 * Poll kie.ai for task completion
 * Uses recordInfo endpoint - same for all models
 * state: waiting/processing = retry, success/done = extract URL, failed = abort
 */
async pollTask(taskId: string): Promise<{ resultUrl: string }> {
  const apiKey = this.getApiKey();

  return pRetry(
    async () => {
      const response = await fetch(
        `https://api.kie.ai/api/v1/playground/recordInfo?taskId=${taskId}`,
        {
          method: 'GET',
          headers: {
            'Authorization': `Bearer ${apiKey}`,
          },
        }
      );

      if (!response.ok) {
        throw new Error(`Query task failed (${response.status}): ${response.statusText}`);
      }

      const data = await response.json();
      const { state, resultJson, failMsg, failCode } = data.data;

      // Still processing - retry
      if (state === 'waiting' || state === 'processing') {
        throw new Error(`Task ${taskId} still ${state}`);
      }

      // Failed - abort
      if (state === 'failed') {
        throw new AbortError(`Task ${taskId} failed: ${failMsg || failCode || 'Unknown'}`);
      }

      // Success - extract URL
      if (state === 'success' || state === 'done') {
        const resultUrl = this.extractResultUrl(resultJson, data.data);
        if (!resultUrl) {
          throw new AbortError(`Task ${taskId} completed but no result URL found`);
        }
        return { resultUrl };
      }

      throw new AbortError(`Task ${taskId} has unknown state: ${state}`);
    },
    {
      retries: 120,  // ~10 minutes of polling
      factor: 1.1,
      minTimeout: 3000,
      maxTimeout: 15000,
      randomize: true,
      onFailedAttempt: (error) => {
        if (error.attemptNumber % 10 === 0) {
          console.log(`[NanoBanana] pollTask attempt ${error.attemptNumber}. ${error.retriesLeft} retries left.`);
        }
      },
    }
  );
}

/**
 * Extract result URL from kie.ai response
 * Handles multiple response formats
 */
private extractResultUrl(resultJson: string | null, data: Record<string, unknown>): string | undefined {
  // Try parsing resultJson
  if (resultJson) {
    try {
      const parsed = JSON.parse(resultJson);
      const url = this.findUrl(parsed);
      if (url) return url;
    } catch {
      // resultJson might be direct URL
      if (typeof resultJson === 'string' && resultJson.startsWith('http')) {
        return resultJson;
      }
    }
  }

  // Fallback: search entire data object
  return this.findUrl(data);
}

/**
 * Recursively find URL in object
 */
private findUrl(obj: unknown, depth = 0): string | undefined {
  if (depth > 5) return undefined;

  if (typeof obj === 'string' && obj.startsWith('http')) {
    return obj;
  }

  if (Array.isArray(obj)) {
    for (const item of obj) {
      const url = this.findUrl(item, depth + 1);
      if (url) return url;
    }
  }

  if (obj && typeof obj === 'object') {
    for (const value of Object.values(obj)) {
      const url = this.findUrl(value, depth + 1);
      if (url) return url;
    }
  }

  return undefined;
}
