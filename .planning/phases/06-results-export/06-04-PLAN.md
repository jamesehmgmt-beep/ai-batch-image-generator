const [job, setJob] = useState<{ outputFormat?: string } | null>(null);

useEffect(() => {
  async function loadData() {
    // Existing generations fetch
    const genRes = await fetch(`/api/job/${jobId}/generations`);
    // ...

    // Also fetch job for format
    const jobRes = await fetch(`/api/job/${jobId}`);
    if (jobRes.ok) {
      const jobData = await jobRes.json();
      setJob({
        outputFormat: jobData.job?.parsed_job?.job?.outputFormat || 'PNG'
      });
    }
  }
  loadData();
}, [jobId]);
