```typescript
// Process each folder
for (const folder of parsedJob.job.folders) {
  const folderPath = folder.folderPath;
  const fileUrls = filesByFolder[folderPath] || [];

  // NEW CASE: Per-image operations specified
  // Cast folder to access imageOperations
  const folderWithImages = folder as typeof folder & {
    imageOperations?: Array<{
      fileName: string;
      model?: string;
      operation?: string;
      resolution?: string;
      quality?: 'basic' | 'high';
      imageSize?: string;
    }>;
  };

  if (folderWithImages.imageOperations && folderWithImages.imageOperations.length > 0) {
    // Process ONLY the files listed in imageOperations
    for (const imgOp of folderWithImages.imageOperations) {
      // Find the file URL matching this fileName
      const matchingUrl = fileUrls.find(url => {
        const urlFileName = url.split('/').pop() || '';
        return urlFileName === imgOp.fileName || urlFileName.toLowerCase() === imgOp.fileName.toLowerCase();
      });

      if (!matchingUrl) {
        console.warn(`[JobManager] imageOperation fileName "${imgOp.fileName}" not found in folder "${folderPath}" - skipping`);
        continue;
      }

      const generationId = uuidv4();

      // Use imageOperation's model, fallback to folder model
      const imageModel: ModelId = (imgOp.model as ModelId) || (folder as any).model || DEFAULT_MODEL;
      const imageStrategy = getModelStrategy(imageModel);
      const maxRefs = imageStrategy.capabilities.maxReferenceImages;

      // Use imageOperation's operation, fallback to folder operation
      const imageOperation = imgOp.operation || buildFinalPrompt(
        jobWithMode.globalPrompt,
        folder.operation,
        combinationMode
      );

      // Combine source file URL with additional reference URLs
      const allReferenceUrls = [matchingUrl, ...additionalReferenceUrls].slice(0, maxRefs);

      const generationJob: GenerationJob = {
        id: generationId,
        jobId,
        folderPath,
        operation: imageOperation,
        model: imageModel,
        // Use imageOp settings if specified, fallback to folder settings
        resolution: imgOp.resolution || folder.resolution,
        aspectRatio: folder.aspectRatio,
        photoMode: folder.photoMode,
        outputFormat: (jobWithMode.outputFormat as 'PNG' | 'JPG') || 'PNG',
        referenceImageUrls: allReferenceUrls,
        sourceFileName: imgOp.fileName,
        quality: imgOp.quality || folder.quality,
        imageSize: imgOp.imageSize || folder.imageSize,
      };

      generationJobs.push(generationJob);

      generationRecords.push({
        id: generationId,
        job_id: jobId,
        folder_path: folderPath,
        operation: imageOperation,
        state: 'pending',
        source_file_name: imgOp.fileName,
        reference_image_urls: allReferenceUrls,
        model: imageModel,
        resolution: imgOp.resolution || folder.resolution,
        aspect_ratio: folder.aspectRatio,
        photo_mode: folder.photoMode,
        quality: imgOp.quality || folder.quality,
        image_size: imgOp.imageSize || folder.imageSize,
        retry_count: 0,
      });
    }

    // Skip the rest of folder processing - imageOperations is exhaustive
    continue;
  }
```
