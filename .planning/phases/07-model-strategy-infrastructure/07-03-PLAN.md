import pRetry, { AbortError } from 'p-retry';
import {
  ModelStrategy,
  ModelCapabilities,
  ModelGenerationParams,
  SeedreamParams,
  SEEDREAM_CAPABILITIES,
} from './types';

/**
 * Strategy implementation for Seedream 4.5 Edit model
 * Supports: quality (basic/high), image_size (named sizes), max 14 reference images
 *
 * Key differences from Nano Banana Pro:
 * - Uses 'quality' (basic/high) instead of 'resolution' (1K/2K/4K)
 * - Uses 'image_urls' instead of 'image_input' for reference images
 * - Uses 'image_size' with named values like 'landscape_16_9' instead of ratio format '16:9'
 * - Supports up to 14 reference images (vs Nano's 8)
 * - Model identifier is 'seedream/4.5-edit' (with slash)
 */
export class SeedreamStrategy implements ModelStrategy {
  readonly capabilities: ModelCapabilities = SEEDREAM_CAPABILITIES;

  /**
   * Validate Seedream 4.5 Edit specific parameters
   */
  validateParams(params: ModelGenerationParams & Partial<SeedreamParams>): void {
    // Validate quality
    const quality = (params as SeedreamParams).quality;
    if (!quality || !['basic', 'high'].includes(quality)) {
      throw new Error(
        `Seedream 4.5 Edit requires quality: basic or high. Got: ${quality}`
      );
    }

    // Validate reference image count (14 max for Seedream)
    if (params.referenceImages.length > this.capabilities.maxReferenceImages) {
      throw new Error(
        `Seedream 4.5 Edit supports max ${this.capabilities.maxReferenceImages} reference images. Got: ${params.referenceImages.length}`
      );
    }

    // Validate image_size if provided
    const imageSize = (params as SeedreamParams).imageSize;
    if (imageSize && !this.capabilities.supportedAspectRatios.includes(imageSize)) {
      throw new Error(
        `Seedream 4.5 Edit does not support image_size: ${imageSize}. Supported: ${this.capabilities.supportedAspectRatios.join(', ')}`
      );
    }
  }

  /**
   * Map standard aspect ratio format to Seedream's image_size format
   * Example: '16:9' -> 'landscape_16_9'
   */
  private mapAspectRatioToImageSize(aspectRatio: string): string {
    const mapping: Record<string, string> = {
      '1:1': 'square',
      '4:3': 'landscape_4_3',
      '3:4': 'portrait_4_3',
      '16:9': 'landscape_16_9',
      '9:16': 'portrait_16_9',
      '3:2': 'landscape_3_2',
      '2:3': 'portrait_3_2',
      '21:9': 'landscape_21_9',
      'auto': 'landscape_16_9', // Default for auto
    };

    return mapping[aspectRatio] || 'landscape_16_9';
  }

  // createTask and pollTask will be implemented in Task 2
}

/**
 * Create a kie.ai task for Seedream 4.5 Edit generation
 * Note: Model identifier is 'seedream/4.5-edit' (with slash)
 * Note: Uses 'image_urls' not 'image_input' for references
 * Note: Uses 'image_resolution' for quality param
 */
async createTask(params: SeedreamParams): Promise<string> {
  this.validateParams(params);

  const apiKey = this.getApiKey();

  // Map aspectRatio to Seedream's image_size format if not already in Seedream format
  const imageSize = params.imageSize || this.mapAspectRatioToImageSize(params.aspectRatio);

  const payload = {
    model: 'seedream/4.5-edit' as const,  // Note: slash in model name
    input: {
      prompt: params.prompt,
      image_urls: params.referenceImages.slice(0, this.capabilities.maxReferenceImages),  // Note: image_urls not image_input
      image_size: imageSize,
      image_resolution: params.quality,  // Note: quality maps to image_resolution
    },
  };

  return pRetry(
    async () => {
      const response = await fetch('https://api.kie.ai/api/v1/jobs/createTask', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${apiKey}`,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(payload),
      });

      // Non-retryable errors
      if ([401, 402, 422].includes(response.status)) {
        const errorText = await response.text();
        throw new AbortError(`Non-retryable error (${response.status}): ${errorText}`);
      }

      // Retryable errors
      if (response.status === 429 || response.status >= 500) {
        throw new Error(`Retryable error (${response.status}): ${response.statusText}`);
      }

      if (!response.ok) {
        const errorText = await response.text();
        throw new AbortError(`API error (${response.status}): ${errorText}`);
      }

      const data = await response.json();
      console.log('[Seedream] createTask response:', JSON.stringify(data, null, 2));

      if (!data.data?.taskId) {
        throw new AbortError(`Invalid response: missing taskId`);
      }

      return data.data.taskId;
    },
    {
      retries: 5,
      factor: 2,
      minTimeout: 1000,
      maxTimeout: 30000,
      randomize: true,
      onFailedAttempt: (error) => {
        console.log(`[Seedream] createTask attempt ${error.attemptNumber} failed. ${error.retriesLeft} retries left.`);
      },
    }
  );
}

/**
 * Poll kie.ai for task completion
 * Same endpoint as Nano Banana - kie.ai recordInfo is model-agnostic
 */
async pollTask(taskId: string): Promise<{ resultUrl: string }> {
  const apiKey = this.getApiKey();

  return pRetry(
    async () => {
      const response = await fetch(
        `https://api.kie.ai/api/v1/playground/recordInfo?taskId=${taskId}`,
        {
          method: 'GET',
          headers: {
            'Authorization': `Bearer ${apiKey}`,
          },
        }
      );

      if (!response.ok) {
        throw new Error(`Query task failed (${response.status}): ${response.statusText}`);
      }

      const data = await response.json();
      const { state, resultJson, failMsg, failCode } = data.data;

      // Still processing - retry
      if (state === 'waiting' || state === 'processing') {
        throw new Error(`Task ${taskId} still ${state}`);
      }

      // Failed - abort
      if (state === 'failed') {
        throw new AbortError(`Task ${taskId} failed: ${failMsg || failCode || 'Unknown'}`);
      }

      // Success - extract URL
      if (state === 'success' || state === 'done') {
        const resultUrl = this.extractResultUrl(resultJson, data.data);
        if (!resultUrl) {
          throw new AbortError(`Task ${taskId} completed but no result URL found`);
        }
        return { resultUrl };
      }

      throw new AbortError(`Task ${taskId} has unknown state: ${state}`);
    },
    {
      retries: 120,
      factor: 1.1,
      minTimeout: 3000,
      maxTimeout: 15000,
      randomize: true,
      onFailedAttempt: (error) => {
        if (error.attemptNumber % 10 === 0) {
          console.log(`[Seedream] pollTask attempt ${error.attemptNumber}. ${error.retriesLeft} retries left.`);
        }
      },
    }
  );
}

private getApiKey(): string {
  const apiKey = process.env.KIE_API_KEY;
  if (!apiKey) {
    throw new Error('KIE_API_KEY environment variable is required');
  }
  return apiKey;
}

private extractResultUrl(resultJson: string | null, data: Record<string, unknown>): string | undefined {
  if (resultJson) {
    try {
      const parsed = JSON.parse(resultJson);
      const url = this.findUrl(parsed);
      if (url) return url;
    } catch {
      if (typeof resultJson === 'string' && resultJson.startsWith('http')) {
        return resultJson;
      }
    }
  }
  return this.findUrl(data);
}

private findUrl(obj: unknown, depth = 0): string | undefined {
  if (depth > 5) return undefined;

  if (typeof obj === 'string' && obj.startsWith('http')) {
    return obj;
  }

  if (Array.isArray(obj)) {
    for (const item of obj) {
      const url = this.findUrl(item, depth + 1);
      if (url) return url;
    }
  }

  if (obj && typeof obj === 'object') {
    for (const value of Object.values(obj)) {
      const url = this.findUrl(value, depth + 1);
      if (url) return url;
    }
  }

  return undefined;
}
